cmake_minimum_required(VERSION 3.14)
project(scalesim_hbm2_replay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SCALE_SIM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(RAMULATOR2_ROOT "${SCALE_SIM_ROOT}/submodules/ramulator2" CACHE PATH "Pinned Ramulator2 source and built library")
find_library(RAMULATOR2_LIBRARY NAMES ramulator PATHS "${RAMULATOR2_ROOT}" NO_DEFAULT_PATH)
if(NOT RAMULATOR2_LIBRARY)
  message(FATAL_ERROR "Run setup_ramulator2.sh to build the pinned Ramulator2 shared library first")
endif()
add_executable(ramulator_hbm2 ramulator_hbm2.cpp)
target_include_directories(ramulator_hbm2 SYSTEM PRIVATE "${RAMULATOR2_ROOT}/src" "${RAMULATOR2_ROOT}/ext/fmt/include")
target_link_libraries(ramulator_hbm2 PRIVATE "${RAMULATOR2_LIBRARY}")
set_target_properties(ramulator_hbm2 PROPERTIES BUILD_RPATH "${RAMULATOR2_ROOT}")
target_compile_options(ramulator_hbm2 PRIVATE -Wall -Wextra)
